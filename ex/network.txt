DATE TIME (SERVER)


#include<stdio.h>
#include<string.h>
#include<unistd.h>
#include<stdlib.h>
#include<sys/types.h>
#include<sys/socket.h>
#include<sys/stat.h>
#include<fcntl.h>
#include<sys/un.h>
#define MAXSIZE 100
#define PATH "/tmp/mysock"
int main()
{
  int sd,bv,lv,av,r,dat;
  char buffer[100];
  socklen_t l;
  struct sockaddr_un myaddr;
  unlink(PATH);
  sd=socket(AF_UNIX,SOCK_STREAM,0);
  if(sd==-1)
  {
    perror("socket_err");
    exit(1);
  }
  myaddr.sun_family=AF_UNIX;
  strncpy(myaddr.sun_path,PATH,sizeof(PATH));
  l=sizeof(myaddr);
  bv=bind(sd,(struct sockaddr*)&myaddr,l);
  if(bv==-1)
  {
   perror("bind_err");
   exit(2);
  }
  lv=listen(sd,10);
  if(lv==-1)
  {
    perror("listen_err");
    exit(3);
  }
while(1)
 {
  printf("waiting for client request-----\n");
  av=accept(sd,(struct sockaddr*)&myaddr,&l);
  if(av==-1)
  {
    perror("accept_err");
    exit(4);
  }
  memset(buffer,'\0',MAXSIZE);
  r=read(av,buffer,MAXSIZE);
  system("date>abc.txt");
  dat=open("abc.txt",O_RDONLY);
  r=read(dat,buffer,MAXSIZE);
  write(av,buffer,r);
  close(av);
 }
return 0;
}

{
int sd,bv,lv,av,r,n;

socklen_t l;

unlink(PATH);

{
